cmake_minimum_required(VERSION 3.22)
project(PhasePocket VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PHASE_DSP_ONLY "Build portable DSP tests without JUCE" OFF)
option(PHASE_BUILD_TESTS "Build integration tests and UI capture" ON)
option(PHASE_ENABLE_OPENGL "Enable optional OpenGL (off at runtime by default)" ON)
option(PHASE_ENABLE_LICENSING "Enable activation for the new Phase product" OFF)
option(PHASE_ENABLE_CLAP "Build CLAP" OFF)
set(PHASE_LICENSE_API_URL "" CACHE STRING "Phase Pocket activation endpoint")
if(PHASE_ENABLE_LICENSING AND NOT PHASE_LICENSE_API_URL MATCHES "^https://[A-Za-z0-9.-]+(:[0-9]+)?(/[A-Za-z0-9/_-]*)?$")
 message(FATAL_ERROR "Configure the Phase product and set its HTTPS activation endpoint before enabling licensing")
endif()
find_package(Threads REQUIRED)
add_library(PhaseDSP STATIC Source/PhaseOptimizer.cpp Source/PhaseOptimizer.h)
set_target_properties(PhaseDSP PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(PhaseDSP PUBLIC Threads::Threads)
target_include_directories(PhaseDSP PUBLIC Source)
include(CTest)
if(BUILD_TESTING)
 add_executable(PhaseDSPTests Tests/dsp_test.cpp)
 target_link_libraries(PhaseDSPTests PRIVATE PhaseDSP)
 add_test(NAME PhaseDSP COMMAND PhaseDSPTests)
endif()
if(PHASE_DSP_ONLY)
 return()
endif()
if(NOT JUCE_DIR)
 set(JUCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../JUCE")
endif()
if(NOT EXISTS "${JUCE_DIR}/CMakeLists.txt")
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
 FetchContent_MakeAvailable(JUCE)
else()
 if(WIN32)
  include(cmake/ApplyWindowsVBlankFix.cmake)
  duck_apply_windows_vblank_fix("${JUCE_DIR}")
 endif()
 add_subdirectory("${JUCE_DIR}" JUCE)
endif()
set(PHASE_DEFAULT_FORMATS VST3 AAX Standalone)
if(APPLE)
 list(APPEND PHASE_DEFAULT_FORMATS AU)
endif()
set(PHASE_FORMATS "${PHASE_DEFAULT_FORMATS}" CACHE STRING "Formats")
juce_add_plugin(PhasePocket COMPANY_NAME "Rainline Music" COMPANY_WEBSITE "https://rainlinemusic.su" COMPANY_EMAIL "contact@rainlinemusic.su" BUNDLE_ID "ru.rainlinemusic.phasepocket" AAX_IDENTIFIER "ru.rainlinemusic.phasepocket" PLUGIN_MANUFACTURER_CODE RnLn PLUGIN_CODE PhPk FORMATS ${PHASE_FORMATS} PRODUCT_NAME "Phase Pocket" DESCRIPTION "Selective phase peak optimizer" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Dynamics AAX_CATEGORY Dynamics)
juce_generate_juce_header(PhasePocket)
configure_file(Source/LicenseConfig.h.in generated/LicenseConfig.h @ONLY)
target_include_directories(PhasePocket PUBLIC "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_sources(PhasePocket PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/PluginProcessor.h Source/PluginEditor.h Source/UIStyle.h Source/License.h Source/LicensePublicKey.h Source/GlowRenderer.h Source/SoftwareGlow.h)
target_compile_definitions(PhasePocket PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0 PHASE_ENABLE_LICENSING=$<BOOL:${PHASE_ENABLE_LICENSING}> DUCK_ENABLE_OPENGL=$<BOOL:${PHASE_ENABLE_OPENGL}> $<$<BOOL:${WIN32}>:NOMINMAX=1> $<$<BOOL:${WIN32}>:WIN32_LEAN_AND_MEAN=1>)
if(MSVC)
 target_compile_options(PhasePocket PRIVATE /utf-8)
else()
 target_compile_options(PhasePocket PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
juce_add_binary_data(PhaseAssets SOURCES Assets/Fonts/Inter-Regular.ttf Assets/Fonts/Inter-Medium.ttf Assets/Designer/DialLarge.png Assets/Designer/DialSmall.png)
set_target_properties(PhaseAssets PROPERTIES POSITION_INDEPENDENT_CODE TRUE)
target_link_libraries(PhasePocket PRIVATE PhaseDSP juce::juce_audio_utils juce::juce_cryptography PUBLIC PhaseAssets juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(PHASE_ENABLE_OPENGL)
 target_link_libraries(PhasePocket PUBLIC juce::juce_opengl)
endif()
if(PHASE_ENABLE_CLAP)
 include(FetchContent)
 FetchContent_Declare(clap_juce_extensions GIT_REPOSITORY https://github.com/free-audio/clap-juce-extensions.git GIT_TAG 7adee3a1bd4684d4caa5601100e364abccff4b4d GIT_SUBMODULES_RECURSE TRUE)
 FetchContent_MakeAvailable(clap_juce_extensions)
 clap_juce_extensions_plugin(TARGET PhasePocket CLAP_ID "ru.rainlinemusic.phasepocket" CLAP_FEATURES audio-effect stereo)
endif()
if(PHASE_BUILD_TESTS)
 juce_add_console_app(PhaseIntegrationTests PRODUCT_NAME "Phase Pocket integration tests")
 juce_generate_juce_header(PhaseIntegrationTests)
 target_sources(PhaseIntegrationTests PRIVATE Tests/integration_test.cpp)
 target_include_directories(PhaseIntegrationTests PRIVATE Source "${CMAKE_CURRENT_BINARY_DIR}/PhasePocket_artefacts/JuceLibraryCode")
 target_compile_definitions(PhaseIntegrationTests PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1)
 target_link_libraries(PhaseIntegrationTests PRIVATE PhasePocket PhaseDSP juce::juce_audio_utils juce::juce_cryptography)
 if(PHASE_ENABLE_OPENGL)
  target_link_libraries(PhaseIntegrationTests PRIVATE juce::juce_opengl)
 endif()
 add_test(NAME PhaseIntegration COMMAND PhaseIntegrationTests)
endif()
foreach(format VST3 AAX AU)
 if(TARGET PhasePocket_${format})
  add_custom_command(TARGET PhasePocket_${format} POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:PhasePocket_${format}>/../Resources" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "$<TARGET_FILE_DIR:PhasePocket_${format}>/../Resources/Inter-OFL.txt" VERBATIM)
 endif()
endforeach()
if(TARGET PhasePocket_CLAP)
 if(APPLE)
  set(PHASE_LICENSE_DIR "$<TARGET_FILE_DIR:PhasePocket_CLAP>/../Resources/Licenses")
 else()
  set(PHASE_LICENSE_DIR "$<TARGET_FILE_DIR:PhasePocket_CLAP>/Licenses")
 endif()
 add_custom_command(TARGET PhasePocket_CLAP POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "${PHASE_LICENSE_DIR}" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Licenses" "${PHASE_LICENSE_DIR}" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/Assets/Fonts/Inter-OFL.txt" "${PHASE_LICENSE_DIR}/Inter-OFL.txt" VERBATIM)
endif()
